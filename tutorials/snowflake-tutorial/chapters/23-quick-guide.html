23. Quick Guide

Snowflake SQL Cheatsheet

CategorySQL
WarehouseCREATE WAREHOUSE wh WAREHOUSE_SIZE = 'XSMALL' AUTO_SUSPEND = 300;
DatabaseCREATE DATABASE db;
CREATE DATABASE db2 CLONE db;
SchemaCREATE SCHEMA db.schema;
TableCREATE TABLE t (id INT, name STRING);
StageCREATE STAGE s;
PUT file://data.csv @s;
LoadCOPY INTO t FROM @s FILE_FORMAT=(TYPE=CSV);
UnloadCOPY INTO @s/out/ FROM t;
GET @s/out/ file://local/;
ExternalCREATE STAGE s3 URL='s3://bucket/' STORAGE_INTEGRATION=i;
Time TravelSELECT * FROM t AT(TIMESTAMP=>...);
CloneCREATE TABLE t2 CLONE t;
