16. Useful Queries

Time Travel

-- Query at a specific timestamp
SELECT * FROM my_table
  AT(TIMESTAMP => '2025-01-01 12:00:00'::TIMESTAMP);

-- Query before a DML statement
SELECT * FROM my_table
  BEFORE(STATEMENT => '019abc123...');

-- Create clone from history
CREATE TABLE restored_table CLONE my_table
  AT(OFFSET => -3600);  -- 1 hour ago

Query Metadata

-- Show tables
SHOW TABLES IN SCHEMA my_db.public;

-- Describe table
DESC TABLE my_table;

-- Query history
SELECT * FROM TABLE(INFORMATION_SCHEMA.QUERY_HISTORY())
  WHERE QUERY_TYPE = 'SELECT'
  ORDER BY START_TIME DESC
  LIMIT 10;

Sampling

-- Random sample (0.1%)
SELECT * FROM my_table SAMPLE (0.1);

-- Row count approximation
SELECT APPROX_COUNT_DISTINCT(customer_id) FROM sales;
