1. Provisioners

Provisioners execute actions on the local machine or on a remote resource during creation or destruction. Terraform's documentation explicitly warns: "Provisioners are a last resort." Prefer user-data, cloud-init, or configuration management tools whenever possible.

+--------------------------------------------------------------+
|                    Provisioner Types                          |
+--------------+------------------+----------------------------+
|  local-exec  |  remote-exec     |  file                       |
|  Runs a      |  SSH/WinRM into  |  Copies files to or from    |
|  command on  |  the target and  |  a remote resource           |
|  the machine |  runs commands   |                              |
|  running     |                  |                              |
|  Terraform   |                  |                              |
+--------------+------------------+----------------------------+

1.1 local-exec

local-exec invokes a local executable after a resource is created. Common use cases include calling Ansible, triggering a script, or writing to a local file.

resource "azurerm_linux_web_app" "example" {
  name                = "my-webapp"
  resource_group_name = azurerm_resource_group.example.name
  location            = azurerm_resource_group.example.location
  service_plan_id     = azurerm_service_plan.example.id

  site_config {}

  provisioner "local-exec" {
    command = "az webapp deploy --resource-group  --name  --src-path ./app.zip"
    environment = {
      ARM_SUBSCRIPTION_ID = data.azurerm_client_config.current.subscription_id
    }
  }

  provisioner "local-exec" {
    when    = destroy
    command = "echo 'WebApp  is being destroyed' >> destroy-log.txt"
  }
}

Key points:

  • self references the resource being provisioned — available inside provisioner and connection blocks.
  • when = destroy runs the provisioner only when the resource is destroyed.
  • interpreter defaults to ["/bin/sh", "-c"] — override for PowerShell or other shells.
  • environment sets environment variables for the command.

1.2 remote-exec

remote-exec connects to a remote resource over SSH or WinRM and executes commands inline.

resource "aws_instance" "web" {
  ami           = "ami-0c55b159cbfafe1f0"
  instance_type = "t3.micro"
  key_name      = aws_key_pair.deployer.key_name

  provisioner "remote-exec" {
    inline = [
      "sudo apt-get update -y",
      "sudo apt-get install -y nginx",
      "sudo systemctl enable nginx",
      "sudo systemctl start nginx",
      "echo 'Nginx installed and running' > /tmp/setup-complete",
    ]

    connection {
      type        = "ssh"
      host        = self.public_ip
      user        = "ubuntu"
      private_key = file("~/.ssh/id_rsa")
      timeout     = "5m"
    }
  }
}

1.3 file provisioner

The file provisioner uploads or downloads files between the Terraform machine and the resource.

resource "aws_instance" "app" {
  ami           = "ami-0c55b159cbfafe1f0"
  instance_type = "t3.micro"

  provisioner "file" {
    source      = "scripts/init.sh"
    destination = "/tmp/init.sh"

    connection {
      type        = "ssh"
      host        = self.public_ip
      user        = "ubuntu"
      private_key = file("~/.ssh/id_rsa")
    }
  }

  provisioner "file" {
    content     = templatefile("templates/nginx.conf.tpl", { port = 8080 })
    destination = "/etc/nginx/nginx.conf"

    connection {
      type        = "ssh"
      host        = self.public_ip
      user        = "ubuntu"
      private_key = file("~/.ssh/id_rsa")
    }
  }
}
Warning: Provisioners make Terraform graphs less predictable. If a provisioner fails, the resource may be left in a half-configured state. Prefer user_data, cloud-init, Ansible, or Packer for configuration management.

2. Lifecycle Rules Deep Dive

The lifecycle block modifies how Terraform creates, updates, and destroys individual resources. Mastering lifecycle rules is essential for zero-downtime deployments and safe infrastructure changes.

2.1 create_before_destroy

Creates the new resource before destroying the old one — critical for load balancers, DNS records, and any resource that must remain available during replacement.

resource "aws_launch_template" "app" {
  name_prefix   = "app-"
  image_id      = data.aws_ami.ubuntu.id
  instance_type = "t3.medium"

  lifecycle {
    create_before_destroy = true
  }
}

2.2 prevent_destroy

Prevents Terraform from destroying the resource. If a terraform destroy or a recreate would destroy this resource, Terraform errors out. Ideal for databases, S3 buckets with data, and critical shared infrastructure.

resource "aws_db_instance" "production" {
  identifier     = "prod-db"
  engine         = "mysql"
  instance_class = "db.r5.large"

  lifecycle {
    prevent_destroy = true
  }
}

2.3 ignore_changes

Tells Terraform to ignore specific attributes when calculating a diff. This prevents Terraform from reverting external changes (e.g., autoscaling, manual console edits, or values managed by another tool).

resource "aws_autoscaling_group" "web" {
  desired_capacity = 2
  min_size         = 1
  max_size         = 5

  lifecycle {
    ignore_changes = [
      desired_capacity,   # Autoscaler manages this
      load_balancers,     # Managed externally
    ]
  }
}

# Ignore all tags (e.g., when tags are injected by a pipeline)
resource "azurerm_linux_virtual_machine" "example" {
  # ... config ...

  lifecycle {
    ignore_changes = [
      tags,          # Ignore all tag changes
      tags["Team"],  # Ignore only a specific tag
    ]
  }
}

2.4 replace_triggered_by

Triggers replacement of the resource when the specified resource or attribute changes. Introduced in Terraform 1.2. This is cleaner than using null_resource with triggers.

resource "aws_instance" "web" {
  ami           = data.aws_ami.latest.id
  instance_type = "t3.micro"

  lifecycle {
    replace_triggered_by = [
      aws_launch_template.app,              # Replace when the whole resource changes
      aws_launch_template.app.image_id,     # Replace when only image_id changes
    ]
  }
}

resource "aws_launch_template" "app" {
  name_prefix   = "app-"
  image_id      = data.aws_ami.ubuntu.id
  instance_type = "t3.medium"
}

2.5 precondition and postcondition

Conditionals that run custom validation logic before and after the resource is planned/applied. If the condition fails, Terraform halts the operation. Use these to enforce invariants that variable validation cannot express.

resource "azurerm_linux_web_app" "api" {
  name                = "api-"
  resource_group_name = azurerm_resource_group.example.name
  location            = azurerm_resource_group.example.location
  service_plan_id     = azurerm_service_plan.example.id

  site_config {}

  lifecycle {
    precondition {
      condition     = var.environment != "production" || var.instance_sku != "B1"
      error_message = "Production environments must use at least an S1 App Service Plan."
    }

    postcondition {
      condition     = self.default_site_hostname != null
      error_message = "Web app created but default_site_hostname is null — deployment may have failed."
    }
  }
}
+------------------------------------------------------+
|                Lifecycle Decision Flow                |
+------------------------------------------------------+
|                                                      |
|  Plan phase:                                         |
|    |-- Has config changed?                           |
|    |    |-- Yes --> check ignore_changes --> diff?    |
|    |    |              |-- ignored --> no change      |
|    |    |              |-- not ignored --> update/rep |
|    |    +-- No --> no change                         |
|    |-- replace_triggered_by?                         |
|    |    +-- trigger changed? --> force replacement   |
|    +-- prevent_destroy + destroying? --> ERROR        |
|                                                      |
|  Apply phase:                                        |
|    |-- create_before_destroy?                        |
|    |    +-- create new --> verify new --> destroy old |
|    |-- precondition fails? --> ERROR                 |
|    +-- postcondition fails? --> ERROR                |
+------------------------------------------------------+

2.6 Combining Multiple Lifecycle Rules

You can combine multiple lifecycle arguments in a single block. Here is a production database example:

resource "aws_db_instance" "main" {
  identifier           = "main-db"
  engine               = "postgres"
  engine_version       = "14.9"
  instance_class       = "db.r6g.xlarge"
  allocated_storage    = 100
  skip_final_snapshot  = false
  final_snapshot_identifier = "main-db-final-"

  lifecycle {
    prevent_destroy = true

    ignore_changes = [
      allocated_storage,  # Auto-scaling handles this
    ]

    precondition {
      condition     = var.environment == "production"
      error_message = "This database should only be created in production."
    }

    postcondition {
      condition     = self.status == "available"
      error_message = "Database did not reach 'available' status."
    }
  }
}

2.7 Lifecycle Rules Reference Table

Rule Description Use Case
create_before_destroy Create replacement before destroying original Zero-downtime replacements, ASG launch configs
prevent_destroy Error if destroy is attempted Databases, critical storage, shared resources
ignore_changes Skip specific attributes during diff External scaling, manual tags, user_data
replace_triggered_by Force replacement when dependency changes Instance refresh on AMI change
precondition Validate before plan/apply Environment-specific constraints
postcondition Validate after create/update Verify resource reached expected state

3. Dynamic Blocks

Dynamic blocks generate repetitive nested blocks from a list or map, eliminating copy-paste boilerplate. They are especially useful for security groups, ingress/egress rules, and complex provider-specific nested attributes.

3.1 Basic Syntax

dynamic "ingress" {
  for_each = var.ingress_rules
  content {
    description = ingress.value.description
    from_port   = ingress.value.port
    to_port     = ingress.value.port
    protocol    = ingress.value.protocol
    cidr_blocks = ingress.value.cidr_blocks
  }
}

3.2 Full Example — Security Group with Dynamic Rules

variable "ingress_rules" {
  description = "Map of ingress rule configurations"
  type = map(object({
    port        = number
    protocol    = string
    cidr_blocks = list(string)
    description = string
  }))
  default = {
    http = {
      port        = 80
      protocol    = "tcp"
      cidr_blocks = ["0.0.0.0/0"]
      description = "HTTP from anywhere"
    }
    https = {
      port        = 443
      protocol    = "tcp"
      cidr_blocks = ["0.0.0.0/0"]
      description = "HTTPS from anywhere"
    }
    ssh = {
      port        = 22
      protocol    = "tcp"
      cidr_blocks = ["10.0.0.0/8"]
      description = "SSH from internal network"
    }
  }
}

resource "aws_security_group" "web" {
  name        = "web-sg"
  description = "Security group for web servers"
  vpc_id      = aws_vpc.main.id

  dynamic "ingress" {
    for_each = var.ingress_rules
    content {
      description = ingress.value.description
      from_port   = ingress.value.port
      to_port     = ingress.value.port
      protocol    = ingress.value.protocol
      cidr_blocks = ingress.value.cidr_blocks
    }
  }

  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
    description = "Allow all outbound"
  }

  tags = {
    Name = "web-sg"
  }
}

3.3 Filtering with Conditional Expressions

You can use for_each with a filtered expression inside a dynamic block to conditionally include blocks:

variable "enable_monitoring" { type = bool }
variable "allowed_cidrs"     { type = list(string) }

resource "azurerm_network_security_group" "example" {
  name                = "example-nsg"
  location            = azurerm_resource_group.example.location
  resource_group_name = azurerm_resource_group.example.name

  # Always included
  security_rule {
    name                       = "allow-https"
    priority                   = 100
    direction                  = "Inbound"
    access                     = "Allow"
    protocol                   = "Tcp"
    source_port_range          = "*"
    destination_port_range     = "443"
    source_address_prefixes    = var.allowed_cidrs
    destination_address_prefix = "*"
  }

  # Conditionally included
  dynamic "security_rule" {
    for_each = var.enable_monitoring ? [1] : []
    content {
      name                       = "allow-monitoring"
      priority                   = 200
      direction                  = "Inbound"
      access                     = "Allow"
      protocol                   = "Tcp"
      source_port_range          = "*"
      destination_port_range     = "9090"
      source_address_prefix      = "10.0.0.0/8"
      destination_address_prefix = "*"
    }
  }

  tags = { Environment = "production" }
}

3.4 Multiple Dynamic Blocks

You can have multiple dynamic blocks in the same resource, each iterating over different collections:

variable "ingress_rules" { type = list(map(any)) }
variable "egress_rules"  { type = list(map(any)) }

resource "aws_security_group" "multi" {
  name   = "multi-sg"
  vpc_id = aws_vpc.main.id

  dynamic "ingress" {
    for_each = var.ingress_rules
    content {
      from_port   = ingress.value["from_port"]
      to_port     = ingress.value["to_port"]
      protocol    = ingress.value["protocol"]
      cidr_blocks = ingress.value["cidr_blocks"]
    }
  }

  dynamic "egress" {
    for_each = var.egress_rules
    content {
      from_port   = egress.value["from_port"]
      to_port     = egress.value["to_port"]
      protocol    = egress.value["protocol"]
      cidr_blocks = egress.value["cidr_blocks"]
    }
  }
}
Tip: dynamic blocks can hurt readability when overused. If your configuration has more than 3-4 dynamic blocks in a resource, consider extracting the logic into a module or using a map-of-objects variable with a single for_each on the resource itself.

4. for_each with Maps and Sets

The for_each meta-argument lets you create multiple instances of a resource or module from a map or set. It is more flexible than count because each instance has a stable, unique key.

4.1 Basic for_each with a Map

variable "regions" {
  type = map(object({
    location          = string
    sku_name          = string
    zone_redundant    = bool
  }))
  default = {
    east = {
      location       = "eastus"
      sku_name       = "B1"
      zone_redundant = false
    }
    west = {
      location       = "westus2"
      sku_name       = "S1"
      zone_redundant = true
    }
  }
}

resource "azurerm_service_plan" "multi_region" {
  for_each = var.regions

  name                = "asp-"
  resource_group_name = azurerm_resource_group.example.name
  location            = each.value.location
  os_type             = "Linux"
  sku_name            = each.value.sku_name
  zone_redundant      = each.value.zone_redundant

  tags = {
    Environment = var.environment
    Region      = each.key
  }
}

output "service_plan_ids" {
  value = { for k, v in azurerm_service_plan.multi_region : k => v.id }
}

4.2 for_each with a Set

When using a set, each element becomes the key. Sets are unordered and best used when you only need a simple list of values:

variable "subnets" {
  type    = set(string)
  default = ["10.0.1.0/24", "10.0.2.0/24", "10.0.3.0/24"]
}

resource "aws_subnet" "private" {
  for_each = var.subnets

  vpc_id            = aws_vpc.main.id
  cidr_block        = each.value
  availability_zone = ""

  tags = {
    Name = "private-"
  }
}

4.3 for_each with a Transformed Collection

Use a for expression to transform or filter collections before passing them to for_each:

resource "azurerm_resource_group" "env" {
  for_each = { for env in var.environments : env.name => env if env.enabled }

  name     = "rg--"
  location = each.value.location

  tags = {
    Environment = each.key
    Project     = var.project
  }
}

4.4 for_each vs count

Feature count for_each
Indexing Numeric (0, 1, 2…) String key (stable identity)
Remove from middle Shifts all subsequent indices (destroys) Other instances unaffected
Input type Number or list Map or set of strings
Conditional creation count = var.enabled ? 1 : 0 for_each = var.enabled ? {x = 1} : {}
Recommended Simple conditionals only Prefer for most multi-instance patterns

4.5 Referencing for_each Instances

# Single instance
azurerm_service_plan.multi_region["east"]

# All instances (splat)
azurerm_service_plan.multi_region[*].id

# In a for expression
[ for k, v in azurerm_service_plan.multi_region : v.id ]

5. Conditional Resource Creation

Terraform does not have an if statement at the resource level. Instead, you use a few established patterns to conditionally create resources.

5.1 Pattern 1 — count with ternary

The simplest approach: set count to 0 or 1 based on a condition.

variable "create_database" {
  type    = bool
  default = true
}

resource "aws_db_instance" "main" {
  count = var.create_database ? 1 : 0

  identifier     = "main-db"
  engine         = "postgres"
  instance_class = "db.t3.micro"
  allocated_storage = 20

  skip_final_snapshot = true
}

# Reference with index
output "db_endpoint" {
  value = var.create_database ? aws_db_instance.main[0].endpoint : "N/A"
}

5.2 Pattern 2 — for_each with conditional map

Better than count when the resource might be combined with other for_each patterns. This avoids the index-shifting problem.

variable "environments" {
  type = map(object({
    enabled = bool
    sku     = string
  }))
}

resource "azurerm_service_plan" "env" {
  for_each = { for k, v in var.environments : k => v if v.enabled }

  name                = "asp-"
  resource_group_name = azurerm_resource_group.example.name
  location            = var.location
  os_type             = "Linux"
  sku_name            = each.value.sku
}

5.3 Pattern 3 — for_each on a filtered list

variable "features" {
  type = list(object({
    name    = string
    enabled = bool
  }))
}

resource "azurerm_app_service_slot" "staging" {
  for_each = { for f in var.features : f.name => f if f.enabled }

  name           = "staging-"
  app_service_id = azurerm_linux_web_app.main.id
  site_config {}
}

5.4 Pattern 4 — count on modules

module "staging" {
  source   = "./modules/web-app"
  count    = var.deploy_staging ? 1 : 0

  app_name = "myapp-staging"
  sku      = "B1"
}

module "production" {
  source   = "./modules/web-app"
  count    = var.deploy_production ? 1 : 0

  app_name = "myapp-prod"
  sku      = "S1"
}

5.5 Pattern 5 — one() for single-instance resources

The one() function safely unwraps a single-element list or returns null if empty, avoiding the verbosity of index references:

output "db_endpoint" {
  value = one(aws_db_instance.main[*].endpoint)
}

output "db_port" {
  value = one(aws_db_instance.main[*].port)
}
Best Practice: Prefer for_each over count for conditional resources. With count, removing an item from the middle of a list shifts all subsequent indices, causing Terraform to destroy and recreate resources unnecessarily.

6. Custom Provider Development Basics

When no existing provider meets your needs, you can build your own. Terraform providers are written in Go and use the HashiCorp Terraform Plugin SDK (or the newer Terraform Plugin Framework).

6.3 provider.go — Provider Definition

package provider

import (
    "context"
    "github.com/hashicorp/terraform-plugin-sdk/v2/diag"
    "github.com/hashicorp/terraform-plugin-sdk/v2/helper/schema"
)

func New() *schema.Provider {
    return &schema.Provider{
        Schema: map[string]*schema.Schema{
            "api_key": {
                Type:        schema.TypeString,
                Required:    true,
                DefaultFunc: schema.EnvDefaultFunc("EXAMPLE_API_KEY", nil),
                Description: "API key for the Example service",
                Sensitive:   true,
            },
            "base_url": {
                Type:        schema.TypeString,
                Optional:    true,
                Default:     "https://api.example.com/v1",
                Description: "Base URL for the API",
            },
        },
        ResourcesMap: map[string]*schema.Schema{
            "example_widget": resourceWidget(),
        },
        DataSourcesMap: map[string]*schema.Schema{
            "example_widget": dataSourceWidget(),
        },
        ConfigureContextFunc: providerConfigure,
    }
}

func providerConfigure(ctx context.Context, d *schema.ResourceData) (interface{}, diag.Diagnostics) {
    apiKey := d.Get("api_key").(string)
    baseURL := d.Get("base_url").(string)

    client, err := newClient(apiKey, baseURL)
    if err != nil {
        return nil, diag.FromErr(err)
    }
    return client, nil
}

6.4 resource_example.go — CRUD Operations

package provider

import (
    "context"
    "github.com/hashicorp/terraform-plugin-sdk/v2/diag"
    "github.com/hashicorp/terraform-plugin-sdk/v2/helper/schema"
)

func resourceWidget() *schema.Resource {
    return &schema.Resource{
        Description:   "Manages a Widget in the Example service",
        CreateContext: resourceWidgetCreate,
        ReadContext:   resourceWidgetRead,
        UpdateContext: resourceWidgetUpdate,
        DeleteContext: resourceWidgetDelete,

        Schema: map[string]*schema.Schema{
            "name": {
                Type:        schema.TypeString,
                Required:    true,
                Description: "Name of the widget",
            },
            "size": {
                Type:        schema.TypeString,
                Optional:    true,
                Default:     "medium",
                Description: "Size: small, medium, or large",
            },
            "arn": {
                Type:        schema.TypeString,
                Computed:    true,
                Description: "ARN of the widget",
            },
        },
    }
}

func resourceWidgetCreate(ctx context.Context, d *schema.ResourceData, meta interface{}) diag.Diagnostics {
    client := meta.(*ExampleClient)
    name := d.Get("name").(string)
    size := d.Get("size").(string)

    widget, err := client.CreateWidget(name, size)
    if err != nil {
        return diag.FromErr(err)
    }

    d.SetId(widget.ID)
    d.Set("arn", widget.ARN)
    return resourceWidgetRead(ctx, d, meta)
}

func resourceWidgetRead(ctx context.Context, d *schema.ResourceData, meta interface{}) diag.Diagnostics {
    client := meta.(*ExampleClient)
    widget, err := client.GetWidget(d.Id())
    if err != nil {
        d.SetId("")
        return nil  // Resource removed outside Terraform
    }
    d.Set("name", widget.Name)
    d.Set("size", widget.Size)
    d.Set("arn", widget.ARN)
    return nil
}

func resourceWidgetUpdate(ctx context.Context, d *schema.ResourceData, meta interface{}) diag.Diagnostics {
    client := meta.(*ExampleClient)
    if d.HasChange("size") {
        err := client.UpdateWidget(d.Id(), d.Get("size").(string))
        if err != nil {
            return diag.FromErr(err)
        }
    }
    return resourceWidgetRead(ctx, d, meta)
}

func resourceWidgetDelete(ctx context.Context, d *schema.ResourceData, meta interface{}) diag.Diagnostics {
    client := meta.(*ExampleClient)
    err := client.DeleteWidget(d.Id())
    if err != nil {
        return diag.FromErr(err)
    }
    return nil
}

7. Terraform Cloud and Terraform Enterprise

5. Conditional Resource Creation

Terraform does not have an if statement at the resource level. Instead, you use a few established patterns to conditionally create resources.

5.1 Pattern 1 — count with ternary

The simplest approach: set count to 0 or 1 based on a condition.

variable "create_database" {
  type    = bool
  default = true
}

resource "aws_db_instance" "main" {
  count = var.create_database ? 1 : 0

  identifier     = "main-db"
  engine         = "postgres"
  instance_class = "db.t3.micro"
  allocated_storage = 20

  skip_final_snapshot = true
}

# Reference with index
output "db_endpoint" {
  value = var.create_database ? aws_db_instance.main[0].endpoint : "N/A"
}

5.2 Pattern 2 — for_each with conditional map

Better than count when the resource might be combined with other for_each patterns. This avoids the index-shifting problem.

variable "environments" {
  type = map(object({
    enabled = bool
    sku     = string
  }))
}

resource "azurerm_service_plan" "env" {
  for_each = { for k, v in var.environments : k => v if v.enabled }

  name                = "asp-"
  resource_group_name = azurerm_resource_group.example.name
  location            = var.location
  os_type             = "Linux"
  sku_name            = each.value.sku
}

5.3 Pattern 3 — for_each on a filtered list

variable "features" {
  type = list(object({
    name    = string
    enabled = bool
  }))
}

resource "azurerm_app_service_slot" "staging" {
  for_each = { for f in var.features : f.name => f if f.enabled }

  name           = "staging-"
  app_service_id = azurerm_linux_web_app.main.id
  site_config {}
}

5.4 Pattern 4 — count on modules

module "staging" {
  source   = "./modules/web-app"
  count    = var.deploy_staging ? 1 : 0

  app_name = "myapp-staging"
  sku      = "B1"
}

module "production" {
  source   = "./modules/web-app"
  count    = var.deploy_production ? 1 : 0

  app_name = "myapp-prod"
  sku      = "S1"
}

5.5 Pattern 5 — one() for single-instance resources

The one() function safely unwraps a single-element list or returns null if empty, avoiding the verbosity of index references:

output "db_endpoint" {
  value = one(aws_db_instance.main[*].endpoint)
}

output "db_port" {
  value = one(aws_db_instance.main[*].port)
}
Best Practice: Prefer for_each over count for conditional resources. With count, removing an item from the middle of a list shifts all subsequent indices, causing Terraform to destroy and recreate resources unnecessarily.

6. Custom Provider Development Basics

When no existing provider meets your needs, you can build your own. Terraform providers are written in Go and use the HashiCorp Terraform Plugin SDK (or the newer Terraform Plugin Framework).

6.3 provider.go — Provider Definition

package provider

import (
    "context"
    "github.com/hashicorp/terraform-plugin-sdk/v2/diag"
    "github.com/hashicorp/terraform-plugin-sdk/v2/helper/schema"
)

func New() *schema.Provider {
    return &schema.Provider{
        Schema: map[string]*schema.Schema{
            "api_key": {
                Type:        schema.TypeString,
                Required:    true,
                DefaultFunc: schema.EnvDefaultFunc("EXAMPLE_API_KEY", nil),
                Description: "API key for the Example service",
                Sensitive:   true,
            },
            "base_url": {
                Type:        schema.TypeString,
                Optional:    true,
                Default:     "https://api.example.com/v1",
                Description: "Base URL for the API",
            },
        },
        ResourcesMap: map[string]*schema.Schema{
            "example_widget": resourceWidget(),
        },
        DataSourcesMap: map[string]*schema.Schema{
            "example_widget": dataSourceWidget(),
        },
        ConfigureContextFunc: providerConfigure,
    }
}

func providerConfigure(ctx context.Context, d *schema.ResourceData) (interface{}, diag.Diagnostics) {
    apiKey := d.Get("api_key").(string)
    baseURL := d.Get("base_url").(string)

    client, err := newClient(apiKey, baseURL)
    if err != nil {
        return nil, diag.FromErr(err)
    }
    return client, nil
}

6.4 resource_example.go — CRUD Operations

package provider

import (
    "context"
    "github.com/hashicorp/terraform-plugin-sdk/v2/diag"
    "github.com/hashicorp/terraform-plugin-sdk/v2/helper/schema"
)

func resourceWidget() *schema.Resource {
    return &schema.Resource{
        Description:   "Manages a Widget in the Example service",
        CreateContext: resourceWidgetCreate,
        ReadContext:   resourceWidgetRead,
        UpdateContext: resourceWidgetUpdate,
        DeleteContext: resourceWidgetDelete,

        Schema: map[string]*schema.Schema{
            "name": {
                Type:        schema.TypeString,
                Required:    true,
                Description: "Name of the widget",
            },
            "size": {
                Type:        schema.TypeString,
                Optional:    true,
                Default:     "medium",
                Description: "Size: small, medium, or large",
            },
            "arn": {
                Type:        schema.TypeString,
                Computed:    true,
                Description: "ARN of the widget",
            },
        },
    }
}

func resourceWidgetCreate(ctx context.Context, d *schema.ResourceData, meta interface{}) diag.Diagnostics {
    client := meta.(*ExampleClient)
    name := d.Get("name").(string)
    size := d.Get("size").(string)

    widget, err := client.CreateWidget(name, size)
    if err != nil {
        return diag.FromErr(err)
    }

    d.SetId(widget.ID)
    d.Set("arn", widget.ARN)
    return resourceWidgetRead(ctx, d, meta)
}

func resourceWidgetRead(ctx context.Context, d *schema.ResourceData, meta interface{}) diag.Diagnostics {
    client := meta.(*ExampleClient)
    widget, err := client.GetWidget(d.Id())
    if err != nil {
        d.SetId("")
        return nil  # Resource removed outside Terraform
    }
    d.Set("name", widget.Name)
    d.Set("size", widget.Size)
    d.Set("arn", widget.ARN)
    return nil
}

func resourceWidgetUpdate(ctx context.Context, d *schema.ResourceData, meta interface{}) diag.Diagnostics {
    client := meta.(*ExampleClient)
    if d.HasChange("size") {
        err := client.UpdateWidget(d.Id(), d.Get("size").(string))
        if err != nil {
            return diag.FromErr(err)
        }
    }
    return resourceWidgetRead(ctx, d, meta)
}

func resourceWidgetDelete(ctx context.Context, d *schema.ResourceData, meta interface{}) diag.Diagnostics {
    client := meta.(*ExampleClient)
    err := client.DeleteWidget(d.Id())
    if err != nil {
        return diag.FromErr(err)
    }
    return nil
}
