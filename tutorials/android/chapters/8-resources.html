Resources & Assets

Resource Types

Android resources are non-code files organized in the res/ directory. They are accessed via the R class.

DirectoryPurpose
res/drawable/Images, shapes, vector drawables, selectors
res/values/Strings, colors, dimensions, themes, styles
res/layout/XML layout files
res/mipmap/Launcher icons (multiple densities)
res/font/Custom font files (TTF, OTF)
res/raw/Raw files (audio, video, JSON)

String Resources

<!-- res/values/strings.xml -->
<resources>
    <string name="app_name">My Android App</string>
    <string name="welcome_message">Welcome, %1$s!</string>
    <string name="item_count">%d items remaining</string>
    <string-array name="months">
        <item>January</item>
        <item>February</item>
        <item>March</item>
    </string-array>
</resources>

// Access in code
getString(R.string.welcome_message, "Alice")
resources.getStringArray(R.array.months)

// Access in XML
android:text="@string/welcome_message"
android:text="@string/welcome_message"

Color Resources

<!-- res/values/colors.xml -->
<resources>
    <color name="primary">#FF6200EE</color>
    <color name="primaryDark">#FF3700B3</color>
    <color name="accent">#FF03DAC5</color>
    <color name="background">#FFFFFFFF</color>
    <color name="textPrimary">#FF1C1B1F</color>
    <color name="textSecondary">#FF49454F</color>
</resources>

// Access
ContextCompat.getColor(this, R.color.primary)
android:background="@color/background"

Dimens Resources

<!-- res/values/dimens.xml -->
<resources>
    <dimen name="padding_small">8dp</dimen>
    <dimen name="padding_medium">16dp</dimen>
    <dimen name="padding_large">24dp</dimen>
    <dimen name="text_size_title">20sp</dimen>
    <dimen name="text_size_body">14sp</dimen>
    <dimen name="elevation_card">4dp</dimen>
</resources>

// Access
resources.getDimension(R.dimen.padding_medium)
android:padding="@dimen/padding_medium"

Raw Resources

// res/raw/config.json
InputStream stream = resources.openRawResource(R.raw.config);
BufferedReader reader = new BufferedReader(new InputStreamReader(stream));
val jsonString = reader.readText()

// Access in code
R.raw.sound_effect  // Returns resource ID
